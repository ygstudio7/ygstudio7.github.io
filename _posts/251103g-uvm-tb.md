# Testbench Top Example

```cpp
module tb_top;
  import uvm_pkg::*;

  bit clk;
  always #10 clk <= ~clk;

  dut_if         dut_if1(clk);
  dut_wrapper    dut_wr0(._if(dut_if1));

  initial begin
    uvm_config_db #(virtual dut_if)::set (null, "uvm_test_top", "dut_if", dut_if1);
    run_test ("base_test");
  end

  initial begin
    $dumpvars;
    $dumpfile("dump.vcd");
  end
endmodul
```

# run_test method

* run_test("baes_test") 는 uvm 시뮬레이션을 시작하는 핵심 진입점

* 지정된 uvm 테스트 클래스를 생성 (instance) 하고,

* 전체 phasing 메커니즘의 실행을 관리

```cpp
// uvm_global.svh
task run_test(string test_name="");
  uvm_root top;
  uvm_coreservice_t cs;
  cs = uvm_coreservice_t::get();
  top = cs.get_root();
  top.run_test(test_name);
endtask
```

마지막 toprun_test()는 아래와 같다

```cpp
task uvm_root::run_test(string test_name="");
  uvm_report_server l_rs;
  uvm_coreservice_t cs = uvm_coreservice_t::get();

  test_name_count = clp.get_arg_values("+UVM_TESTNAME=", test_names);

  if(test_name_count>0) begin
    test_name = test_names[0];
    testname_plusarg = 1;
  end

  fork begin
    phase_runner_proc = process:self();
    uvm_phase::m_run_phases();
  end
  join_none // 위와 관련없이 바로 다음 실행

  wait(m_phase_all_done==1);

  phase_runner_proc.kill();

  l_rs = uvm_report_server::get_server();
  l_rs.report_summarize();

  if(finish_on-completion)
    $finish;
endtask


```

# Clock generation

기존 legacy tb는 아래와 같이 구성했음

(-) 주파수, 듀티, 위상 같이 어려가지 동적으로 업데이트가 필요할 수 있다 -> 좀더 잘되는 방식이 있으면 좋다

```verilog
// Module level clock generation
module clk_main (...);

 	// Code for clock generation

endmodule

// Instantiated and connected to an interface
module tb_top;
	bit clk_main_out;

	clk_main u_clk_main_0 ( .out (clk_main_out),
							...
						  );

	dut_if 	dut_if0 ( clk_main_out, ...);
endmodule
```



(+) uvm에서는 에이전트 구성 객체를 사용하여, sequence 및 test 에서 clock을 쉽게 제어할 수 있도록 클럭 agent를 생성할 수 있다

```verilog
class clk_agent extents uvm_agents;
  clk_cfg m_clk_cfg;

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
  endfunction

endclass


class base_test extends uvm_tests;
  clk_cfg m_clk_cfg;

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    
    m_clk_cfg.m_freq = 500;
  endfunction

endclass

```

# Internal TAP points

* tb에서 디버깅이나 모니터링을 위해 내부 TAP (Test Access Point)을 생성할 수 있다

* 값을 강제로 변경하거나, 샘플링하기 위해

* 이를 위한 특별 agent를 구성할 수 도 있다



아래 gen_if는 내부 신호를 net으로 설정한 100-bit TAP points들이다

```verilog
interface gen_if;
	logic [99:0] 	signals; 	// General 100-bit wide vector

endinterface

module tb_top;
	gen_if 	u_if0 ();

	des 	u_des 	( ... );

	// Assign an internal net to a generic interface signal
	assign u_if0.signals[23] = u_des.u_xyz.u_abc.status;

endmodule
```
