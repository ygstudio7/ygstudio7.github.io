# Sequence item

* reg_item 은 주소, write data를 저장하도록 정의
* driver로 전송전에 sequence 내부에서 무직위로 지정
* gen_random은 seqr에서 randomize()를 사용할 수 없을때, 직접 호출할 수 있다.

```verilog
// -------------------------------------------------
// reg_item.svh
// This is the base transaction object that will be used
// in the environment to initiate new transactions and 
// capture transactions at DUT interface
class reg_item extends uvm_sequence_item;
  rand bit [ADDR_WIDTH-1:0]  	addr;
  rand bit [DATA_WIDTH-1:0] 	wdata;
  rand bit 						wr;
  bit [DATA_WIDTH-1:0] 		rdata;  

  // Use utility macros to implement standard functions
  // like print, copy, clone, etc
  `uvm_object_utils_begin(reg_item)
  	`uvm_field_int (addr, UVM_DEFAULT)
  	`uvm_field_int (wdata, UVM_DEFAULT)
  	`uvm_field_int (rdata, UVM_DEFAULT)
  	`uvm_field_int (wr, UVM_DEFAULT)
  `uvm_object_utils_end

  function void gen_random();
    addr = $urandom_range(0,2**ADDR_WIDTH-1);
    wdata = $urandom_range(0,2**DATA_WIDTH-1);
    wr = $urandom_range(0,1);    
  endfunction

  virtual function string convert2str();
    return $sformatf("addr=0x%0h wr=0x%0h wdata=0x%0h rdata=0x%0h", addr, wr, wdata, rdata);
  endfunction
  
  function new(string name = "reg_item");
    super.new(name);
  endfunction
endclass
```



# uvm sequencer

* 테스트 벤치의 stimulus 및 랜점화 기능을 담당
* sequence가 sequencer에서 시작되면, **body()**메서드가 실행되어, 궁극적으로 주소/데이터를 랜덤화해서 드라이버로 전송한다
* randomize()를 실행할 수 없을때, gen_random()을 직접 구현해서 호출할 수 있다. 
* reg_item::type_id:;create()호출후, start_item(), finish_item()이 들어가서 하나씩 내보내게 된다

```verilog
// -------------------------------------------------
// gen_item_seq.svh
// 여러가지 data item을 조합해서, 다양한 sequence를 만든다. sequencer에 의해 실행되어 item이 drv에 전달됨
class gen_item_seq extends uvm_sequence;
  `uvm_object_utils(gen_item_seq)
  function new(string name="gen_item_seq");
    super.new(name);
  endfunction
  
  rand int num; 	// Config total number of items to be sent
  
  constraint c1 { soft num inside {[2:5]}; }
  
  virtual task body();
    for (int i = 0; i < num; i ++) begin
    	reg_item m_item = reg_item::type_id::create("m_item");
    	start_item(m_item);
    	//m_item.randomize();
      m_item.gen_random();
      `uvm_info("SEQ", $sformatf("Generate new item: %s", m_item.convert2str()), UVM_LOW)
      	finish_item(m_item);
    end
    `uvm_info("SEQ", $sformatf("Done generation of %0d items", num), UVM_LOW)
  endtask
endclass
```



# uvm driver

* 표준 메서드 run_phase() 에서 주요 동작을 구현한다
* forever 에서, reg_item을 생성하고, seq_item_port.get_next_item()으로 it를 받아오고, drive DUT한 후, seq_item_port.item_done()으로 현재 it를 마무리



```verilog
// -------------------------------------------------
// driver.svh
// sequencer에서 받은 packet을 DUT interface 규격을 맞춰서, DUT 를 drive하는 역할
class driver extends uvm_driver #(reg_item);
  `uvm_component_utils(driver)
  function new(string name = "driver", uvm_component parent=null);
    super.new(name, parent);
  endfunction

  virtual reg_if vif;

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    if (!uvm_config_db#(virtual reg_if)::get(this, "", "reg_vif", vif))
      `uvm_fatal("DRV", "Could not get vif")
  endfunction

  virtual task run_phase(uvm_phase phase);
    super.run_phase(phase);
    forever begin
      reg_item m_item;
      `uvm_info("DRV", $sformatf("Wait for item from sequencer"), UVM_LOW)

      seq_item_port.get_next_item(m_item);
      
      drive_item(m_item); // drive DUT
      
      seq_item_port.item_done();
    end
  endtask

  virtual task drive_item(reg_item m_item);
      vif.sel <= 1;
      vif.addr 	<= m_item.addr;
      vif.wr 	<= m_item.wr;
      vif.wdata <= m_item.wdata;
      
      @ (posedge vif.clk);
      
      while (!vif.ready)  begin
        `uvm_info("DRV", "Wait until ready is high", UVM_LOW)
        @(posedge vif.clk);
      end

      vif.sel <= 0;
  endtask

endclass

```



# uvm_monitor

* uvm_analysis_port를 포함하고, build_phase()에서 생성

* run_phase에서 forever 를 통해, 

  * vif를 보고 있다가,

  * 완전한 transaction이 되면, 

  * ap.write(it)로 analysis_port로 transaction정보  it를 보낸다

    

```verilog
// -------------------------------------------------
// monitor.svh
// DUT 쪽 interface에서 신호 및 bus transaction을 관찰 후, 검증을 위해 상위 컴포넌트인 scoreboard에 전달
class monitor extends uvm_monitor;
  `uvm_component_utils(monitor)

  function new(string name="monitor", uvm_component parent=null);
    super.new(name, parent);
  endfunction

  uvm_analysis_port  #(reg_item) mon_analysis_port;
  
  virtual reg_if vif;
  
  semaphore sema4;

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    if (!uvm_config_db#(virtual reg_if)::get(this, "", "reg_vif", vif))
      `uvm_fatal("MON", "Could not get vif")
    sema4 = new(1);
    mon_analysis_port = new ("mon_analysis_port", this);
  endfunction

  virtual task run_phase(uvm_phase phase);
    super.run_phase(phase);
    // This task monitors the interface for a complete
    // transaction and writes into analysis port when complete
    forever begin
      @ (posedge vif.clk);

      if (vif.sel) begin
        reg_item item = new;
        item.addr = vif.addr;
        item.wr = vif.wr;
        item.wdata = vif.wdata;

        if (!vif.wr) begin
          @(posedge vif.clk);
        	item.rdata = vif.rdata;
        end
      
        `uvm_info(get_type_name(), $sformatf("Monitor found packet %s", item.convert2str()), UVM_LOW)
        mon_analysis_port.write(item);
      end
    end
  endtask
endclass
```



# uvm agent

* build_phase에서 driver, monitor, seqr를 생성한다.
* connect_phase에서 seqr-> drv에 연결한다: seqr의 seq_item_export를 drv의 seq_item_port.connect()로 연결한다. 

```verilog
// -------------------------------------------------
// agent.svh
// sequencer, driver 와 monitor를 포함하는 단일 container으로 입력을 관장한다
class agent extends uvm_agent;
  `uvm_component_utils(agent)
  function new(string name="agent", uvm_component parent=null);
    super.new(name, parent);
  endfunction

  driver 		d0; 		// Driver handle
  monitor 		m0; 		// Monitor handle
  uvm_sequencer #(reg_item)	s0; 		// Sequencer Handle

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    s0 = uvm_sequencer#(reg_item)::type_id::create("s0", this);
    d0 = driver::type_id::create("d0", this);
    m0 = monitor::type_id::create("m0", this);
  endfunction

   // seq를 drv에 연결
  virtual function void connect_phase(uvm_phase phase);
    super.connect_phase(phase);
    d0.seq_item_port.connect(s0.seq_item_export);  // drv.seq_item_port: drv가 transaction을 받는 TLM port
                                                   // s0.seq_item_export: seq가 transaction을 내보내는 TLM export
  endfunction
endclass
```



# uvm scoreboard

* uvm_analysis_imp를 포함하고, build_phase()에서 생성
* 표준 메소드인 write()에서, uvm_analysis_imp를 통해 수신한 transaction을 처리할 작업을 정의
* 
  * 

```verilog
// -------------------------------------------------
// scoreboard.svh
// monitor에서 uvm_analysis_imp를 통해 packet인 reg_item을 받아서,  packet 를 검증하는 역할을 한다
class scoreboard extends uvm_scoreboard;
  `uvm_component_utils(scoreboard)
  function new(string name="scoreboard", uvm_component parent=null);
    super.new(name, parent);
  endfunction

  reg_item refq[DEPTH];

  // reg_item을 받을 수 있도록 TLM analysis port를 만든다
  uvm_analysis_imp #(reg_item, scoreboard) m_analysis_imp;

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    m_analysis_imp = new("m_analysis_imp", this);
  endfunction

  // TLM analysis port에서 reg_item을 받을때 수행할 작업들을 정의
  virtual function write(reg_item item);
     if (item.wr) begin
        if (refq[item.addr] == null)
          refq[item.addr] = new;

        refq[item.addr] = item;
       `uvm_info(get_type_name(), $sformatf("Store addr=0x%0h wr=0x%0h data=0x%0h", item.addr, item.wr, item.wdata), UVM_LOW)
      end

        if (!item.wr) begin
          if (refq[item.addr] == null)
            if (item.rdata != 'h1234)
              `uvm_error (get_type_name(), $sformatf("First time read, addr=0x%0h exp=1234 act=0x%0h",
                                                     item.addr, item.rdata))
          	else
              `uvm_info(get_type_name(), $sformatf("PASS! First time read, addr=0x%0h exp=1234 act=0x%0h",
                                                   item.addr, item.rdata), UVM_LOW)
          else
            if (item.rdata != refq[item.addr].wdata)
              `uvm_error (get_type_name(), $sformatf("addr=0x%0h exp=0x%0h act=0x%0h",
                                                    item.addr, refq[item.addr].wdata, item.rdata))
           else
             `uvm_info(get_type_name(), $sformatf("PASS! addr=0x%0h exp=0x%0h act=0x%0h",
                       item.addr, refq[item.addr].wdata, item.rdata), UVM_LOW)
        end
  endfunction
endclass

```



# uvm env

* 여기에, agent와 scoreboard를 포함하고, 
* connect_phase에서 agent의 monitor를 scoreboard에 analysis_port를 통해 연결한다

```verilog
// -------------------------------------------------
// env.svh
// 모든 uvm components들을 생성하고, 연결하는 역할을 한다
class env extends uvm_env;
  `uvm_component_utils(env)

  function new(string name="env", uvm_component parent=null);
    super.new(name, parent);
  endfunction

  agent 		   a0; 		// Agent handle
  scoreboard	sb0; 		// Scoreboard handle

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    a0 = agent::type_id::create("a0", this);
    sb0 = scoreboard::type_id::create("sb0", this);
  endfunction

  virtual function void connect_phase(uvm_phase phase);
    super.connect_phase(phase);
    a0.m0.mon_analysis_port.connect(sb0.m_analysis_imp);
  endfunction
endclass
```



# uvm test

* env 를 포함하고, build_phase()에서 생성
* run_phase()에서 reset()하고, seq를 생성해서, seq.start(e0.a0.s0)를 호출해 시작하고, seq 동작이 완료되면 다음으로 넘어간다
  * seq는 자신을 동작시킬 sequencer를 e0.a0.s0로 설정한다
  * 여기서 설정한 s0가 seq.body()를 실행시키며, item을 random하게 생성해서, drv로 전달한다. 
    * start_item(it)로 
* raise_objection() ~ drop.objection()은 uvm의 시작과 끝을 알린다. drop.objection()전에 uvm이 종료되는 것을 막기 위함이다

```verilog
// -------------------------------------------------
// test.svh
// 모든 tb 요소들이 포함된 env를 생성하고, 다양한 test case scenario를 여기서 test한다.
class test extends uvm_test;
  `uvm_component_utils(test)
  function new(string name = "test", uvm_component parent=null);
    super.new(name, parent);
  endfunction

  env e0;
  virtual reg_if vif;

  virtual function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    e0 = env::type_id::create("e0", this);
    if (!uvm_config_db#(virtual reg_if)::get(this, "", "reg_vif", vif))
      `uvm_fatal("TEST", "Did not get vif")

      uvm_config_db#(virtual reg_if)::set(this, "e0.a0.*", "reg_vif", vif);
  endfunction

  virtual task run_phase(uvm_phase phase);
    gen_item_seq seq = gen_item_seq::type_id::create("seq");
    phase.raise_objection(this);
    apply_reset();

    //seq.randomize() with {num inside {[20:30]}; };
    seq.num = $urandom_range(20,30);
    seq.start(e0.a0.s0);
    #200;
    phase.drop_objection(this);
  endtask

  virtual task apply_reset();
    vif.rstn <= 0;
    repeat(5) @ (posedge vif.clk);
    vif.rstn <= 1;
    repeat(10) @ (posedge vif.clk);
  endtask
endclass
```



# tb_top

* test와 DUT를 포함하고, 연결한다.
* run_test("test")호출해서 test를 실행한다

```verilog
// tb.sv
`include "uvm_macros.svh"
import uvm_pkg::*;
import env_pkg::*;

// Top level testbench module to instantiate design, interface
// start clocks and run the test
module tb_top;
  reg clk;
  
  always #10 clk =~ clk;
  reg_if _if (clk);
  
  reg_ctrl u0 (   
   .clk     (clk),
   .addr    (_if.addr),
   .rstn    (_if.rstn),
   .sel     (_if.sel),
   .wr      (_if.wr),
   .wdata   (_if.wdata),
   .rdata   (_if.rdata),
   .ready   (_if.ready)
   );
  
  test t0;
  
  initial begin
    clk <= 0;
    uvm_config_db#(virtual reg_if)::set(null, "uvm_test_top", "reg_vif", _if);
    run_test("test");
  end
  
  // System tasks to dump VCD waveform file
  initial begin
    $dumpvars;
    $dumpfile ("dump.vcd");
  end
endmodule

```
