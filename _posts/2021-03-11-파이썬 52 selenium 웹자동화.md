# 참고 내용

import unittest
from selenium import webdriver
import time
from time import sleep

# Importing options for the Opera module
from selenium.webdriver.opera.options import Options

from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.common.by import By

class SeachTest(unittest.TestCase):
    def setUp(self):
        options = Options()
        # Setting the location of the Opera Browser
        options.binary_location = r'C:/Users/ygkim/AppData/Local/Programs/Opera/launcher.exe'
        opera_profile = "e:/_ygkim/ubox/uprofile/ollehbaker"
        options.add_argument("user-data-dir=" + opera_profile)
        # Creation of Opera WebDriver instance
        self.driver = webdriver.Opera(options=options, executable_path=r'C:/Users/ygkim/AppData/Local/Programs/Opera/launcher.exe')

    def test_Search(self):
        driver = self.driver
        driver.maximize_window()
        driver.get("https://www.naver.com/")
        elem = WebDriverWait(driver, 20).until(EC.element_to_be_clickable((By.NAME, "q")))
        elem.send_keys("Lambdatest")
        elem.submit()

        sleep(10)

    def tearDown(self):
        # Close the browser.
        self.driver.close()

if __name__ == '__main__':
    unittest.main()
